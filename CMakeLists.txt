cmake_minimum_required(VERSION 3.21)
project(SoccerBetAdviser VERSION 2.2.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
    add_compile_options(/utf-8 /W3)
else()
    add_compile_options(-Wall -Wextra -Wno-unused-parameter)
endif()

option(SBA_BUILD_UI    "Build the Qt desktop application"            ON)
option(SBA_BUILD_TESTS "Build the core unit tests (no Qt required)"  ON)
option(SBA_DOWNLOAD_ONNXRUNTIME "Download ONNX Runtime automatically when ONNXRUNTIME_ROOT is not set" ON)
set(ONNXRUNTIME_ROOT "" CACHE PATH "Folder of an extracted onnxruntime-<platform>-<version> release")
set(SBA_ORT_VERSION "1.20.1" CACHE STRING "ONNX Runtime version used for the automatic download")
set(SBA_SQLITE_URL "https://www.sqlite.org/2024/sqlite-amalgamation-3450300.zip" CACHE STRING "SQLite amalgamation (used when no system SQLite is found)")

include(FetchContent)

# ---------------------------------------------------------------- nlohmann_json
FetchContent_Declare(nlohmann_json
    URL https://github.com/nlohmann/json/releases/download/v3.12.0/json.tar.xz
    DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(nlohmann_json)

# ---------------------------------------------------------------- SQLite
find_package(SQLite3 QUIET)
if(SQLite3_FOUND AND NOT WIN32)
    add_library(sba_sqlite INTERFACE)
    if(TARGET SQLite3::SQLite3)
        target_link_libraries(sba_sqlite INTERFACE SQLite3::SQLite3)
    else()
        target_link_libraries(sba_sqlite INTERFACE SQLite::SQLite3)
    endif()
else()
    FetchContent_Declare(sqlite_amalgamation URL ${SBA_SQLITE_URL} SOURCE_SUBDIR _none DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
    FetchContent_MakeAvailable(sqlite_amalgamation)
    file(GLOB_RECURSE _sqlite_c "${sqlite_amalgamation_SOURCE_DIR}/sqlite3.c")
    list(GET _sqlite_c 0 _sqlite_src)
    get_filename_component(_sqlite_dir "${_sqlite_src}" DIRECTORY)
    set_source_files_properties("${_sqlite_src}" PROPERTIES LANGUAGE C)
    add_library(sba_sqlite STATIC "${_sqlite_src}")
    target_include_directories(sba_sqlite PUBLIC "${_sqlite_dir}")
    target_compile_definitions(sba_sqlite PRIVATE SQLITE_OMIT_LOAD_EXTENSION)
    if(UNIX)
        target_link_libraries(sba_sqlite PUBLIC ${CMAKE_DL_LIBS} pthread)
    endif()
endif()

# ---------------------------------------------------------------- ONNX Runtime
set(SBA_HAVE_ONNX OFF)
if(NOT ONNXRUNTIME_ROOT AND SBA_DOWNLOAD_ONNXRUNTIME)
    if(WIN32)
        set(_ort_url "https://github.com/microsoft/onnxruntime/releases/download/v${SBA_ORT_VERSION}/onnxruntime-win-x64-${SBA_ORT_VERSION}.zip")
    elseif(APPLE)
        set(_ort_url "https://github.com/microsoft/onnxruntime/releases/download/v${SBA_ORT_VERSION}/onnxruntime-osx-universal2-${SBA_ORT_VERSION}.tgz")
    else()
        set(_ort_url "https://github.com/microsoft/onnxruntime/releases/download/v${SBA_ORT_VERSION}/onnxruntime-linux-x64-${SBA_ORT_VERSION}.tgz")
    endif()
    FetchContent_Declare(onnxruntime URL ${_ort_url} SOURCE_SUBDIR _none DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
    FetchContent_MakeAvailable(onnxruntime)
    set(ONNXRUNTIME_ROOT "${onnxruntime_SOURCE_DIR}")
endif()
if(ONNXRUNTIME_ROOT)
    find_path(SBA_ORT_INCLUDE onnxruntime_cxx_api.h HINTS "${ONNXRUNTIME_ROOT}/include" NO_DEFAULT_PATH)
    find_library(SBA_ORT_LIB onnxruntime HINTS "${ONNXRUNTIME_ROOT}/lib" NO_DEFAULT_PATH)
    if(SBA_ORT_INCLUDE AND SBA_ORT_LIB)
        set(SBA_HAVE_ONNX ON)
        message(STATUS "ONNX Runtime: ${SBA_ORT_LIB}")
    else()
        message(WARNING "ONNXRUNTIME_ROOT is set but headers/library were not found - building WITHOUT the Random Forest.")
    endif()
else()
    message(WARNING "No ONNX Runtime - building WITHOUT the Random Forest (Poisson model only).")
endif()

# ---------------------------------------------------------------- core library (Qt-free)
file(GLOB SBA_CORE_SOURCES CONFIGURE_DEPENDS src/core/*.cpp src/core/*.h)
add_library(sba_core STATIC ${SBA_CORE_SOURCES})
target_include_directories(sba_core PUBLIC src)
target_link_libraries(sba_core PUBLIC nlohmann_json::nlohmann_json sba_sqlite)
if(SBA_HAVE_ONNX)
    target_compile_definitions(sba_core PUBLIC SBA_HAS_ONNX)
    target_include_directories(sba_core PUBLIC "${SBA_ORT_INCLUDE}")
    target_link_libraries(sba_core PUBLIC "${SBA_ORT_LIB}")
    if(UNIX AND NOT APPLE)
        set(CMAKE_BUILD_RPATH "${ONNXRUNTIME_ROOT}/lib")
        set(CMAKE_INSTALL_RPATH "\$ORIGIN")
    endif()
endif()
if(MSVC)
    target_compile_definitions(sba_core PUBLIC _CRT_SECURE_NO_WARNINGS NOMINMAX)
endif()

# Copies model/data files and the ONNX Runtime DLLs next to a built executable.
function(sba_stage_runtime target)
    set(_res "$<TARGET_FILE_DIR:${target}>/resources")
    foreach(f data/EPL_merged_2017-2027.csv models/epl_decision_config.json models/poisson_model.json models/random_forest_epl.onnx)
        add_custom_command(TARGET ${target} POST_BUILD
            COMMAND ${CMAKE_COMMAND} -E make_directory "${_res}/data" "${_res}/models"
            COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_SOURCE_DIR}/ml_training/${f}" "${_res}/${f}")
    endforeach()
    if(SBA_HAVE_ONNX AND WIN32)
        file(GLOB _ort_dlls "${ONNXRUNTIME_ROOT}/lib/*.dll")
        foreach(d ${_ort_dlls})
            add_custom_command(TARGET ${target} POST_BUILD
                COMMAND ${CMAKE_COMMAND} -E copy_if_different "${d}" "$<TARGET_FILE_DIR:${target}>")
        endforeach()
    endif()
endfunction()

# ---------------------------------------------------------------- tests
if(SBA_BUILD_TESTS)
    enable_testing()
    add_executable(sba_tests tests/test_core.cpp)
    target_link_libraries(sba_tests PRIVATE sba_core)
    target_compile_definitions(sba_tests PRIVATE SBA_SOURCE_DIR="${CMAKE_SOURCE_DIR}")
    add_test(NAME core COMMAND sba_tests WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}")
    sba_stage_runtime(sba_tests)
endif()

# ---------------------------------------------------------------- Qt application
if(SBA_BUILD_UI)
    find_package(Qt6 REQUIRED COMPONENTS Widgets Network Concurrent)
    find_package(Qt6 QUIET OPTIONAL_COMPONENTS Svg)
    qt_standard_project_setup()
    file(GLOB SBA_APP_SOURCES CONFIGURE_DEPENDS src/ui/*.cpp src/ui/*.h src/ui/user/*.cpp src/ui/user/*.h src/ui/art/*.cpp src/ui/art/*.h src/ui/design/*.cpp src/ui/design/*.h src/api/*.cpp src/api/*.h)
    qt_add_executable(SoccerBetAdviser WIN32 src/main.cpp ${SBA_APP_SOURCES})
    qt_add_resources(SoccerBetAdviser sba_icons PREFIX "/" BASE "${CMAKE_SOURCE_DIR}/assets"
        FILES assets/icons/chevron-down.svg assets/icons/check.svg assets/icons/chevron-down-dark.svg)
    target_include_directories(SoccerBetAdviser PRIVATE src)
    target_link_libraries(SoccerBetAdviser PRIVATE sba_core Qt6::Widgets Qt6::Network Qt6::Concurrent)
    if(TARGET Qt6::Svg)
        target_link_libraries(SoccerBetAdviser PRIVATE Qt6::Svg)
        target_compile_definitions(SoccerBetAdviser PRIVATE SBA_HAS_SVG=1)
    else()
        message(WARNING "Qt6 Svg module not found: SVG club crests will be skipped (PNG crests and generated crests still work).")
    endif()
    target_compile_definitions(SoccerBetAdviser PRIVATE SBA_SOURCE_DIR="${CMAKE_SOURCE_DIR}" SBA_VERSION="${PROJECT_VERSION}")
    sba_stage_runtime(SoccerBetAdviser)
    if(EXISTS "${CMAKE_SOURCE_DIR}/logos")      # optional club logos shipped next to the executable
        add_custom_command(TARGET SoccerBetAdviser POST_BUILD
            COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_SOURCE_DIR}/logos" "$<TARGET_FILE_DIR:SoccerBetAdviser>/logos")
    endif()

    if(SBA_BUILD_TESTS)
        add_executable(sba_api_tests tests/test_api.cpp src/api/FootballDataClient.cpp src/api/OddsApiClient.cpp src/api/HttpClient.cpp src/api/LiveDataService.cpp src/api/AppSettings.cpp src/api/ResultsUpdater.cpp)
        target_include_directories(sba_api_tests PRIVATE src)
        target_link_libraries(sba_api_tests PRIVATE sba_core Qt6::Core Qt6::Network)
        add_test(NAME api COMMAND sba_api_tests)

        add_executable(sba_design_tests tests/test_design.cpp src/ui/Theme.cpp src/ui/user/UserTheme.cpp)
        target_include_directories(sba_design_tests PRIVATE src src/ui)
        target_link_libraries(sba_design_tests PRIVATE Qt6::Gui)
        add_test(NAME design COMMAND sba_design_tests)
    endif()

    # `cmake --build . --target deploy` gathers the Qt DLLs next to the exe (Windows).
    if(WIN32)
        get_target_property(_qmake_exe Qt6::qmake IMPORTED_LOCATION)
        get_filename_component(_qt_bin "${_qmake_exe}" DIRECTORY)
        find_program(SBA_WINDEPLOYQT windeployqt HINTS "${_qt_bin}")
        if(SBA_WINDEPLOYQT)
            add_custom_target(deploy
                COMMAND "${SBA_WINDEPLOYQT}" --no-translations --no-system-d3d-compiler --no-opengl-sw "$<TARGET_FILE:SoccerBetAdviser>"
                DEPENDS SoccerBetAdviser
                COMMENT "Running windeployqt")
        endif()
    endif()

    include(GNUInstallDirs)
    install(TARGETS SoccerBetAdviser RUNTIME DESTINATION .)
    install(DIRECTORY "$<TARGET_FILE_DIR:SoccerBetAdviser>/resources" DESTINATION .)
    foreach(_doc README.md LICENSE-NOTICES.md)
        if(EXISTS "${CMAKE_SOURCE_DIR}/${_doc}")
            install(FILES ${_doc} DESTINATION .)
        endif()
    endforeach()
    if(EXISTS "${CMAKE_SOURCE_DIR}/logos")
        install(DIRECTORY logos DESTINATION .)
    endif()
    if(SBA_HAVE_ONNX AND WIN32)
        install(DIRECTORY "${ONNXRUNTIME_ROOT}/lib/" DESTINATION . FILES_MATCHING PATTERN "*.dll")
    endif()
endif()
